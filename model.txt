class MaskedConv1d(nn.Conv1d):
    """
    Input:
        x: [B, L, C]
        input_mask: [B, L] or [B, L, 1]
    Output:[B, L, C_out]
    """
    def __init__(
        self,
        in_channels: int,
        out_channels: int,
        kernel_size: int,
        stride: int = 1,
        dilation: int = 1,
        groups: int = 1,
        bias: bool = True,
    ):
        padding = dilation * (kernel_size - 1) // 2
        super().__init__(
            in_channels,
            out_channels,
            kernel_size,
            stride=stride,
            dilation=dilation,
            groups=groups,
            bias=bias,
            padding=padding,
        )

    def forward(self, x, input_mask=None):
        if input_mask is not None:
            if input_mask.dim() == 2:
                input_mask = input_mask.unsqueeze(-1)
            x = x * input_mask.to(x.dtype)
        return super().forward(x.transpose(1, 2)).transpose(1, 2)


class AttnMean(nn.Module):
    """[B, L, H] -> [B, H]
    """
    def __init__(self, hidden_size) -> None:
        super().__init__()
        self.layer_norm = nn.LayerNorm(hidden_size)
        self.layer = MaskedConv1d(hidden_size, 1, 1)

    def forward(self, x, input_mask=None):
        x = self.layer_norm(x)
        batch_size = x.shape[0]

        attn = self.layer(x)                 # [B, L, 1]
        attn = attn.view(batch_size, -1)     # [B, L]

        if input_mask is not None:
            attn = attn.masked_fill_(
                ~input_mask.view(batch_size, -1).bool(), float("-inf")
            )

        attn = F.softmax(attn, dim=-1)
        x = torch.bmm(attn.unsqueeze(1), x).squeeze(1)   #[B, H]
        return x


class AttnTransform(nn.Module):
    """[B, L, H] -> [B, L, H]
    """
    def __init__(self, hidden_size) -> None:
        super().__init__()
        self.hidden_size = hidden_size
        self.layer_norm = nn.LayerNorm(hidden_size)
        self.layer = MaskedConv1d(hidden_size, 1, 1)

    def forward(self, x, input_mask=None):
        x = self.layer_norm(x)
        batch_size = x.shape[0]

        attn = self.layer(x)                 # [B, L, 1]
        attn = attn.view(batch_size, -1)     # [B, L]

        if input_mask is not None:
            attn = attn.masked_fill_(
                ~input_mask.view(batch_size, -1).bool(), float("-inf")
            )

        attn = F.softmax(attn, dim=-1).view(batch_size, -1, 1)
        x = attn * x
        return x


class OutHead(nn.Module):
    def __init__(self, hidden_dim, out_dim):
        super().__init__()
        self.fc1 = nn.Linear(hidden_dim, hidden_dim // 2)
        self.ac = nn.Tanh()
        self.dropout = nn.Dropout(0.1)
        self.fc2 = nn.Linear(hidden_dim // 2, hidden_dim // 2)
        self.re = nn.ReLU()
        self.linear = nn.Linear(hidden_dim // 2, out_dim)

    def forward(self, x):
        x = self.ac(self.fc1(x))
        x = self.dropout(x)
        x = self.re(self.fc2(x))
        x = self.linear(x)
        return x

# fallback sequence mixer if real mamba is not available
# ============================================================
class FallbackSequenceMixer(nn.Module):
    def __init__(self, d_model, expand=2, dropout=0.1):
        super().__init__()
        self.dwconv = nn.Conv1d(
            d_model, d_model, kernel_size=3, padding=1, groups=d_model
        )
        self.pw1 = nn.Linear(d_model, d_model * expand)
        self.act = nn.SiLU()
        self.pw2 = nn.Linear(d_model * expand, d_model)
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        y = self.dwconv(x.transpose(1, 2)).transpose(1, 2)
        y = self.pw2(self.act(self.pw1(y)))
        y = self.dropout(y)
        return y


class ResidualMambaBlock(nn.Module):
    def __init__(self, hidden_size, d_state=16, d_conv=4, expand=2, dropout=0.1):
        super().__init__()
        self.norm = nn.LayerNorm(hidden_size)

        if _HAS_MAMBA:
            self.mixer = Mamba(
                d_model=hidden_size,
                d_state=d_state,
                d_conv=d_conv,
                expand=expand,
            )
        else:
            self.mixer = FallbackSequenceMixer(
                d_model=hidden_size,
                expand=expand,
                dropout=dropout,
            )

        self.dropout = nn.Dropout(dropout)

    def forward(self, x, input_mask=None):
        residual = x
        h = self.norm(x)

        if input_mask is not None:
            if input_mask.dim() == 2:
                input_mask = input_mask.unsqueeze(-1)
            input_mask = input_mask.to(h.dtype)
            h = h * input_mask

        h = self.mixer(h)
        h = self.dropout(h)

        if input_mask is not None:
            h = h * input_mask

        return residual + h


class MambaSequenceBranch(nn.Module):
    """
    initial sequence features -> mamba blocks -> attention pooling -> [B, H]
    """
    def __init__(
        self,
        hidden_size,
        num_layers=2,
        d_state=16,
        d_conv=4,
        expand=2,
        dropout=0.1,
    ):
        super().__init__()
        self.in_norm = nn.LayerNorm(hidden_size)
        self.blocks = nn.ModuleList([
                ResidualMambaBlock(
                    hidden_size=hidden_size,
                    d_state=d_state,
                    d_conv=d_conv,
                    expand=expand,
                    dropout=dropout,
                )
                for _ in range(num_layers)
            ]
        )
        self.out_norm = nn.LayerNorm(hidden_size)
        self.pool = AttnMean(hidden_size)

    def forward(self, x, input_mask=None):
        x = self.in_norm(x)
        for blk in self.blocks:
            x = blk(x, input_mask=input_mask)
        x = self.out_norm(x)
        x = self.pool(x, input_mask=input_mask)   # [B, H]
        return x

class ProjectionHead(nn.Module):
    def __init__(self, in_dim, proj_dim=256, dropout=0.1):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(in_dim, in_dim),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(in_dim, proj_dim),
        )

    def forward(self, x):
        return self.net(x)


class SymmetricInfoNCE(nn.Module):
    """
    z1[i] <-> z2[i] is positive
    z1[i] <-> z2[j], i!=j are negatives
    """
    def __init__(self, temperature=0.1):
        super().__init__()
        self.temperature = temperature

    def forward(self, z1, z2):
        if z1.size(0) <= 1:
            return z1.new_zeros(())

        z1 = F.normalize(z1, dim=-1)
        z2 = F.normalize(z2, dim=-1)

        logits_12 = torch.matmul(z1, z2.transpose(0, 1)) / self.temperature
        logits_21 = torch.matmul(z2, z1.transpose(0, 1)) / self.temperature

        targets = torch.arange(z1.size(0), device=z1.device)
        loss_12 = F.cross_entropy(logits_12, targets)
        loss_21 = F.cross_entropy(logits_21, targets)
        return 0.5 * (loss_12 + loss_21)


class BranchAlignBlock(nn.Module):
    """
    Contrastive Learning I:
    Align main-branch feature and mamba-branch feature
    for the same sample and same feature type.
    """
    def __init__(self, hidden_size, proj_dim=256, temperature=0.1, dropout=0.1):
        super().__init__()
        self.main_refine = nn.Sequential(
            nn.LayerNorm(hidden_size),
            nn.Linear(hidden_size, hidden_size),
            nn.GELU(),
            nn.Dropout(dropout),
        )
        self.mamba_refine = nn.Sequential(
            nn.LayerNorm(hidden_size),
            nn.Linear(hidden_size, hidden_size),
            nn.GELU(),
            nn.Dropout(dropout),
        )

        self.main_proj = ProjectionHead(hidden_size, proj_dim, dropout)
        self.mamba_proj = ProjectionHead(hidden_size, proj_dim, dropout)
        self.loss_fn = SymmetricInfoNCE(temperature)

    def forward(self, main_feat, mamba_feat, compute_loss=True):
        main_feat_1 = main_feat + self.main_refine(main_feat)
        mamba_feat_1 = mamba_feat + self.mamba_refine(mamba_feat)

        if compute_loss:
            cl_loss = self.loss_fn(
                self.main_proj(main_feat_1),
                self.mamba_proj(mamba_feat_1),
            )
        else:
            cl_loss = main_feat.new_zeros(())

        return main_feat_1, mamba_feat_1, cl_loss

# ============================================================
class BilinearFeatureFusion(nn.Module):
    """
    Advanced Bilinear Multiplicative Fusion
    利用哈达玛乘积(Hadamard Product)捕捉序列突变(Mamba)与结构交互(Attention)的高阶特征交叉
    """
    def __init__(self, hidden_size, dropout=0.1):
        super().__init__()
        # 将两者映射到更适合交互的子空间
        self.proj_main = nn.Linear(hidden_size, hidden_size)
        self.proj_mamba = nn.Linear(hidden_size, hidden_size)
        
        # 融合网络：输入包含原始特征(2H)和它们的乘性交互特征(H)，共3H
        self.fuse_net = nn.Sequential(
            nn.Linear(hidden_size * 3, hidden_size),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(hidden_size, hidden_size)
        )
        
        # 动态门控，决定融合特征保留的比例
        self.gate = nn.Sequential(
            nn.Linear(hidden_size * 3, hidden_size),
            nn.Sigmoid()
        )
        
        self.norm = nn.LayerNorm(hidden_size)
        self.dropout = nn.Dropout(dropout)

    def forward(self, main_feat, mamba_feat):
        # 1. 非线性映射
        h_main = F.gelu(self.proj_main(main_feat))
        h_mamba = F.gelu(self.proj_mamba(mamba_feat))
        
        # 2. 核心：哈达玛乘积捕捉特征交叉
        interaction = h_main * h_mamba
        
        # 3. 拼接特征：[B, 3 * H]
        concat_feat = torch.cat([main_feat, mamba_feat, interaction], dim=-1)
        
        # 4. 生成融合候选与门控权重
        candidate = self.fuse_net(concat_feat)
        g = self.gate(concat_feat)
        
        # 5. 残差连接：以主路为基底，动态注入交互信息
        fused = main_feat + g * candidate
        fused = self.norm(fused)
        fused = self.dropout(fused)
        
        return fused


class CrossSampleContrastiveRefiner(nn.Module):
    """
    Contrastive Learning II:
    Use batch-wise instance discrimination on fused features.
    """
    def __init__(self, hidden_size, proj_dim=256, temperature=0.1, dropout=0.1):
        super().__init__()
        self.refine = nn.Sequential(
            nn.LayerNorm(hidden_size),
            nn.Linear(hidden_size, hidden_size),
            nn.GELU(),
            nn.Dropout(dropout),
        )
        self.pre_proj = ProjectionHead(hidden_size, proj_dim, dropout)
        self.post_proj = ProjectionHead(hidden_size, proj_dim, dropout)
        self.loss_fn = SymmetricInfoNCE(temperature)

    def forward(self, feat, compute_loss=True):
        feat_2 = feat + self.refine(feat)

        if compute_loss:
            cl_loss = self.loss_fn(
                self.pre_proj(feat),
                self.post_proj(feat_2),
            )
        else:
            cl_loss = feat.new_zeros(())

        return feat_2, cl_loss


class SeqBindModel(nn.Module):
    def __init__(self, args):
        super(SeqBindModel, self).__init__()
        self.config = args
        self.hidden_size = args.hidden_size
        self.num_heads = args.num_heads
        self.dropout = args.dropout

        # -----------------------------
        # optional hyperparameters
        # -----------------------------
        self.use_mamba_branch = getattr(args, "use_mamba_branch", True)
        self.mamba_layers = getattr(args, "mamba_layers", 2)
        self.mamba_d_state = getattr(args, "mamba_d_state", 16)
        self.mamba_d_conv = getattr(args, "mamba_d_conv", 4)
        self.mamba_expand = getattr(args, "mamba_expand", 2)
        self.mamba_dropout = getattr(args, "mamba_dropout", args.dropout)

        self.use_contrastive = getattr(args, "use_contrastive", True)
        self.proj_dim = getattr(args, "proj_dim", 256)
        self.contrastive_temperature = getattr(args, "contrastive_temperature", 0.1)

        # =============================
        # backbone encoder
        # =============================
        self.encoder = EsmModel.from_pretrained(args.model_locate)

        # =============================
        # original main branch
        # =============================
        self.wt_ab_conv_transformer = AttnTransform(hidden_size=self.hidden_size)
        self.wt_ag_conv_transformer = AttnTransform(hidden_size=self.hidden_size)
        self.mt_ab_conv_transformer = AttnTransform(hidden_size=self.hidden_size)
        self.mt_ag_conv_transformer = AttnTransform(hidden_size=self.hidden_size)

        self.wt_ab_attn = MutilHeadSelfAttn(
            num_heads=self.num_heads, hidden_dim=self.hidden_size, dropout=self.dropout
        )
        self.mt_ab_attn = MutilHeadSelfAttn(
            num_heads=self.num_heads, hidden_dim=self.hidden_size, dropout=self.dropout
        )
        self.wt_ag_attn = MutilHeadSelfAttn(
            num_heads=self.num_heads, hidden_dim=self.hidden_size, dropout=self.dropout
        )
        self.mt_ag_attn = MutilHeadSelfAttn(
            num_heads=self.num_heads, hidden_dim=self.hidden_size, dropout=self.dropout
        )

        self.wt_ab_mean = AttnMean(self.hidden_size)
        self.mt_ab_mean = AttnMean(self.hidden_size)
        self.wt_ag_mean = AttnMean(self.hidden_size)
        self.mt_ag_mean = AttnMean(self.hidden_size)

        # =============================
        # mamba branch
        # =============================
        if self.use_mamba_branch:
            self.wt_ab_mamba_branch = MambaSequenceBranch(
                hidden_size=self.hidden_size,
                num_layers=self.mamba_layers,
                d_state=self.mamba_d_state,
                d_conv=self.mamba_d_conv,
                expand=self.mamba_expand,
                dropout=self.mamba_dropout,
            )
            self.wt_ag_mamba_branch = MambaSequenceBranch(
                hidden_size=self.hidden_size,
                num_layers=self.mamba_layers,
                d_state=self.mamba_d_state,
                d_conv=self.mamba_d_conv,
                expand=self.mamba_expand,
                dropout=self.mamba_dropout,
            )
            self.mt_ab_mamba_branch = MambaSequenceBranch(
                hidden_size=self.hidden_size,
                num_layers=self.mamba_layers,
                d_state=self.mamba_d_state,
                d_conv=self.mamba_d_conv,
                expand=self.mamba_expand,
                dropout=self.mamba_dropout,
            )
            self.mt_ag_mamba_branch = MambaSequenceBranch(
                hidden_size=self.hidden_size,
                num_layers=self.mamba_layers,
                d_state=self.mamba_d_state,
                d_conv=self.mamba_d_conv,
                expand=self.mamba_expand,
                dropout=self.mamba_dropout,
            )

        # =============================
        # Contrastive Learning I
        # =============================
        self.wt_ab_align = BranchAlignBlock(
            hidden_size=self.hidden_size,
            proj_dim=self.proj_dim,
            temperature=self.contrastive_temperature,
            dropout=self.dropout,
        )
        self.wt_ag_align = BranchAlignBlock(
            hidden_size=self.hidden_size,
            proj_dim=self.proj_dim,
            temperature=self.contrastive_temperature,
            dropout=self.dropout,
        )
        self.mt_ab_align = BranchAlignBlock(
            hidden_size=self.hidden_size,
            proj_dim=self.proj_dim,
            temperature=self.contrastive_temperature,
            dropout=self.dropout,
        )
        self.mt_ag_align = BranchAlignBlock(
            hidden_size=self.hidden_size,
            proj_dim=self.proj_dim,
            temperature=self.contrastive_temperature,
            dropout=self.dropout,
        )

        # =============================
        # Fusion (已替换为双线性乘性融合)
        # =============================
        self.wt_ab_fusion = BilinearFeatureFusion(self.hidden_size, dropout=self.dropout)
        self.wt_ag_fusion = BilinearFeatureFusion(self.hidden_size, dropout=self.dropout)
        self.mt_ab_fusion = BilinearFeatureFusion(self.hidden_size, dropout=self.dropout)
        self.mt_ag_fusion = BilinearFeatureFusion(self.hidden_size, dropout=self.dropout)

        # =============================
        # Contrastive Learning II
        # =============================
        self.wt_ab_refiner = CrossSampleContrastiveRefiner(
            hidden_size=self.hidden_size,
            proj_dim=self.proj_dim,
            temperature=self.contrastive_temperature,
            dropout=self.dropout,
        )
        self.wt_ag_refiner = CrossSampleContrastiveRefiner(
            hidden_size=self.hidden_size,
            proj_dim=self.proj_dim,
            temperature=self.contrastive_temperature,
            dropout=self.dropout,
        )
        self.mt_ab_refiner = CrossSampleContrastiveRefiner(
            hidden_size=self.hidden_size,
            proj_dim=self.proj_dim,
            temperature=self.contrastive_temperature,
            dropout=self.dropout,
        )
        self.mt_ag_refiner = CrossSampleContrastiveRefiner(
            hidden_size=self.hidden_size,
            proj_dim=self.proj_dim,
            temperature=self.contrastive_temperature,
            dropout=self.dropout,
        )

        # =============================
        # prediction head 
        # 【修改二】：因为后面使用了 Concat + 差异项构建复合体，维度从 2H 扩展到了 8H
        # =============================
        self.bn = nn.BatchNorm1d(self.hidden_size * 8)
        self.out_head = OutHead(self.hidden_size * 8, 1)

        if self.config.freeze_backbone:
            for param in self.encoder.parameters():
                param.requires_grad = False

    @torch.no_grad()
    def get_encoder_embeddings(self, input_ids, attention_mask):
        return self.encoder(
            input_ids=input_ids,
            attention_mask=attention_mask
        ).last_hidden_state

    def forward(
        self,
        wt_ab_inputs_ids,
        wt_ab_inputs_mask,
        mut_ab_inputs_ids,
        mt_ab_inputs_mask,
        wt_ag_inputs_ids,
        wt_ag_inputs_mask,
        mut_ag_inputs_ids,
        mt_ag_inputs_mask,
        wt_features=None,
        mt_features=None,
        labels=None,
        return_aux=False,
    ):
        # ======================================================
        # 1) ESM encoder outputs
        # ======================================================
        wt_ab_embeddings = self.get_encoder_embeddings(
            input_ids=wt_ab_inputs_ids, attention_mask=wt_ab_inputs_mask
        )  #[B, L, H]
        mt_ab_embeddings = self.get_encoder_embeddings(
            input_ids=mut_ab_inputs_ids, attention_mask=mt_ab_inputs_mask
        )
        wt_ag_embeddings = self.get_encoder_embeddings(
            input_ids=wt_ag_inputs_ids, attention_mask=wt_ag_inputs_mask
        )
        mt_ag_embeddings = self.get_encoder_embeddings(
            input_ids=mut_ag_inputs_ids, attention_mask=mt_ag_inputs_mask
        )

        # keep a copy for mamba branch
        wt_ab_embeddings_init = wt_ab_embeddings
        wt_ag_embeddings_init = wt_ag_embeddings
        mt_ab_embeddings_init = mt_ab_embeddings
        mt_ag_embeddings_init = mt_ag_embeddings

        # ======================================================
        # 2) original main branch
        # ======================================================
        wt_ab_embeddings = self.wt_ab_conv_transformer(
            wt_ab_embeddings, wt_ab_inputs_mask
        )
        mt_ab_embeddings = self.mt_ab_conv_transformer(
            mt_ab_embeddings, mt_ab_inputs_mask
        )
        wt_ag_embeddings = self.wt_ag_conv_transformer(
            wt_ag_embeddings, wt_ag_inputs_mask
        )
        mt_ag_embeddings = self.mt_ag_conv_transformer(
            mt_ag_embeddings, mt_ag_inputs_mask
        )

        # ag looks at ab
        wt_ag_ctx = self.wt_ag_attn(
            wt_ag_embeddings, wt_ab_embeddings, wt_ab_embeddings, mask=wt_ab_inputs_mask
        )
        mt_ag_ctx = self.mt_ag_attn(
            mt_ag_embeddings, mt_ab_embeddings, mt_ab_embeddings, mask=mt_ab_inputs_mask
        )

        # ab looks at ag
        wt_ab_ctx = self.wt_ab_attn(
            wt_ab_embeddings, wt_ag_embeddings, wt_ag_embeddings, mask=wt_ag_inputs_mask
        )
        mt_ab_ctx = self.mt_ab_attn(
            mt_ab_embeddings, mt_ag_embeddings, mt_ag_embeddings, mask=mt_ag_inputs_mask
        )

        # pool main branch -> [B, H]
        main_wt_ag = self.wt_ag_mean(wt_ag_ctx, wt_ag_inputs_mask)
        main_mt_ag = self.mt_ag_mean(mt_ag_ctx, mt_ag_inputs_mask)
        main_wt_ab = self.wt_ab_mean(wt_ab_ctx, wt_ab_inputs_mask)
        main_mt_ab = self.mt_ab_mean(mt_ab_ctx, mt_ag_inputs_mask if False else mt_ab_inputs_mask)

        # ======================================================
        # 3) mamba branch ->[B, H]
        # ======================================================
        if self.use_mamba_branch:
            m_wt_ab = self.wt_ab_mamba_branch(wt_ab_embeddings_init, wt_ab_inputs_mask)
            m_wt_ag = self.wt_ag_mamba_branch(wt_ag_embeddings_init, wt_ag_inputs_mask)
            m_mt_ab = self.mt_ab_mamba_branch(mt_ab_embeddings_init, mt_ab_inputs_mask)
            m_mt_ag = self.mt_ag_mamba_branch(mt_ag_embeddings_init, mt_ag_inputs_mask)
        else:
            # fallback: use zeros if disabled
            m_wt_ab = main_wt_ab.new_zeros(main_wt_ab.shape)
            m_wt_ag = main_wt_ag.new_zeros(main_wt_ag.shape)
            m_mt_ab = main_mt_ab.new_zeros(main_mt_ab.shape)
            m_mt_ag = main_mt_ag.new_zeros(main_mt_ag.shape)

        # ======================================================
        # 4) Contrastive Learning I (branch alignment)
        # ======================================================
        compute_cl = return_aux and self.use_contrastive

        main_wt_ab_1, m_wt_ab_1, cl1_wt_ab = self.wt_ab_align(main_wt_ab, m_wt_ab, compute_loss=compute_cl)
        main_wt_ag_1, m_wt_ag_1, cl1_wt_ag = self.wt_ag_align(main_wt_ag, m_wt_ag, compute_loss=compute_cl)
        main_mt_ab_1, m_mt_ab_1, cl1_mt_ab = self.mt_ab_align(main_mt_ab, m_mt_ab, compute_loss=compute_cl)
        main_mt_ag_1, m_mt_ag_1, cl1_mt_ag = self.mt_ag_align(main_mt_ag, m_mt_ag, compute_loss=compute_cl)

        cl1_loss = 0.25 * (cl1_wt_ab + cl1_wt_ag + cl1_mt_ab + cl1_mt_ag)

        # ======================================================
        # 5) Feature Fusion
        # ======================================================
        fused_wt_ab = self.wt_ab_fusion(main_wt_ab_1, m_wt_ab_1)
        fused_wt_ag = self.wt_ag_fusion(main_wt_ag_1, m_wt_ag_1)
        fused_mt_ab = self.mt_ab_fusion(main_mt_ab_1, m_mt_ab_1)
        fused_mt_ag = self.mt_ag_fusion(main_mt_ag_1, m_mt_ag_1)

        # ======================================================
        # 6) Contrastive Learning II (cross-sample discrimination)
        # ======================================================
        final_wt_ab, cl2_wt_ab = self.wt_ab_refiner(fused_wt_ab, compute_loss=compute_cl)
        final_wt_ag, cl2_wt_ag = self.wt_ag_refiner(fused_wt_ag, compute_loss=compute_cl)
        final_mt_ab, cl2_mt_ab = self.mt_ab_refiner(fused_mt_ab, compute_loss=compute_cl)
        final_mt_ag, cl2_mt_ag = self.mt_ag_refiner(fused_mt_ag, compute_loss=compute_cl)

        cl2_loss = 0.25 * (cl2_wt_ab + cl2_wt_ag + cl2_mt_ab + cl2_mt_ag)

        # ======================================================
        # 7) final prediction
        # 【修改三】：重构复合体特征构建逻辑，用Concat+交互项替换加法
        # ======================================================
        # 隐式表达Ab与Ag的距离、相似度和结合界面信息
        wt_complex = torch.cat([
            final_wt_ab, 
            final_wt_ag, 
            final_wt_ab * final_wt_ag,                # 乘积：局部相互作用强度
            torch.abs(final_wt_ab - final_wt_ag)      # 绝对差：特征差异度
        ], dim=-1) # [B, 4 * H]

        mt_complex = torch.cat([
            final_mt_ab, 
            final_mt_ag, 
            final_mt_ab * final_mt_ag, 
            torch.abs(final_mt_ab - final_mt_ag)
        ], dim=-1) # [B, 4 * H]

        # 拼接野生型与突变型复合体特征用于预测 ΔΔG
        rep = torch.cat([wt_complex, mt_complex], dim=-1)    # [B, 8 * H]
        rep = self.bn(rep)
        pred = self.out_head(rep).squeeze(1)                 # [B]

        if return_aux:
            return {
                "pred": pred,
                "cl1_loss": cl1_loss,
                "cl2_loss": cl2_loss,
                "main_feats": {
                    "wt_ab": main_wt_ab,
                    "wt_ag": main_wt_ag,
                    "mt_ab": main_mt_ab,
                    "mt_ag": main_mt_ag,
                },
                "mamba_feats": {
                    "wt_ab": m_wt_ab,
                    "wt_ag": m_wt_ag,
                    "mt_ab": m_mt_ab,
                    "mt_ag": m_mt_ag,
                },
                "fused_feats": {
                    "wt_ab": final_wt_ab,
                    "wt_ag": final_wt_ag,
                    "mt_ab": final_mt_ab,
                    "mt_ag": final_mt_ag,
                },
            }

        return pred
